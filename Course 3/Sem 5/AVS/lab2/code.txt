#include <stdio.h>
#include <stdlib.h>

int main(int argc, char *argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Использование: %s <путь_к_файлу>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *file_path = argv[1];
    FILE *file = fopen(file_path, "r");

    if (file == NULL) {
        perror("Ошибка открытия файла");
        return EXIT_FAILURE;
    }

    char buffer[1024];
    size_t bytesRead;

    printf("Содержимое файла '%s':\n", file_path);

    while (1) {
        bytesRead = fread(buffer, 1, sizeof(buffer), file);

        if (bytesRead == 0) {
            if (feof(file)) {
                break;
            } else {
                perror("Ошибка чтения файла");
                fclose(file);
                return EXIT_FAILURE;
            }
        }

        fwrite(buffer, 1, bytesRead, stdout);
    }

    fclose(file);

    printf("\nОжидание сигнала остановки...\n");
    while (1) {
    }

    return EXIT_SUCCESS;
}
